cmake_minimum_required(VERSION 3.20)
project(learning_backend LANGUAGES C)
if(CMAKE_C_COMPILER MATCHES "python-zig")
    # Zig is the installed native compiler here; it also supplies its archiver.
    set(CMAKE_C_ARCHIVE_CREATE "\"${CMAKE_C_COMPILER}\" ar qc <TARGET> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_FINISH "")
endif()

if(NOT ESP_GSP_COMPONENT_DIR AND DEFINED ENV{ESP_GSP_COMPONENT_DIR})
    set(ESP_GSP_COMPONENT_DIR "$ENV{ESP_GSP_COMPONENT_DIR}")
endif()
if(NOT ESP_GSP_COMPONENT_DIR)
    foreach(candidate IN ITEMS
            "${CMAKE_CURRENT_LIST_DIR}/../managed_components/espressif__esp-gsp")
        get_filename_component(candidate "${candidate}" ABSOLUTE)
        if(EXISTS "${candidate}/tools/sim_bridge/CMakeLists.txt")
            set(ESP_GSP_COMPONENT_DIR "${candidate}")
            break()
        endif()
    endforeach()
endif()
if(NOT EXISTS "${ESP_GSP_COMPONENT_DIR}/tools/sim_bridge/CMakeLists.txt")
    message(FATAL_ERROR
        "Install espressif/esp-gsp first, or set ESP_GSP_COMPONENT_DIR")
endif()

add_subdirectory("${ESP_GSP_COMPONENT_DIR}/tools/sim_bridge" sim_bridge)
gsp_add_backend(learning_backend
    SOURCES ../main/learning_ui.c ../main/learning_model.c ../main/learning_controls.c platform_pc.c
    INCLUDE_DIRECTORIES ../main
    SCENES ../ui/main.json)
target_compile_definitions(learning_backend PRIVATE
    LEARNING_SIM_PHOTO="${CMAKE_CURRENT_LIST_DIR}/fixtures/question.png")

add_executable(learning_model_test learning_model_test.c ../main/learning_model.c)
add_executable(questions_test questions_test.c ../main/learning_model.c)
target_include_directories(questions_test PRIVATE ../main)
if(MSVC)
    target_compile_options(questions_test PRIVATE /utf-8 /UNDEBUG)
else()
    target_compile_options(questions_test PRIVATE -UNDEBUG)
endif()
add_executable(controls_test controls_test.c ../main/learning_controls.c)
target_include_directories(controls_test PRIVATE ../main)
add_executable(speech_conversation_test speech_conversation_test.c ../main/learning_speech.c ../main/learning_conversation.c)
target_include_directories(speech_conversation_test PRIVATE stubs ../main)
add_executable(orientation_logic_test orientation_logic_test.c ../main/app_orientation_logic.c)
target_include_directories(orientation_logic_test PRIVATE ../main)
add_executable(viewport_test viewport_test.c ../main/learning_viewport.c)
target_include_directories(viewport_test PRIVATE ../main)
target_include_directories(learning_model_test PRIVATE ../main)
add_executable(provider_protocol_test provider_protocol_test.c ../main/learning_provider.c
    ../main/learning_config.c ../main/learning_controls.c ../managed_components/espressif__cjson/cJSON/cJSON.c)
target_include_directories(provider_protocol_test PRIVATE stubs ../main ../managed_components/espressif__cjson/cJSON)
if(MSVC)
    target_compile_options(controls_test PRIVATE /UNDEBUG)
    target_compile_options(speech_conversation_test PRIVATE /utf-8 /UNDEBUG)
    target_compile_options(learning_model_test PRIVATE /utf-8 /UNDEBUG)
    target_compile_options(provider_protocol_test PRIVATE /utf-8 /UNDEBUG)
    target_compile_options(orientation_logic_test PRIVATE /UNDEBUG)
    target_compile_options(viewport_test PRIVATE /UNDEBUG)
else()
    target_compile_options(controls_test PRIVATE -UNDEBUG)
    target_compile_options(speech_conversation_test PRIVATE -UNDEBUG)
    target_compile_options(learning_model_test PRIVATE -UNDEBUG)
    target_compile_options(provider_protocol_test PRIVATE -UNDEBUG)
    target_compile_options(orientation_logic_test PRIVATE -UNDEBUG)
    target_compile_options(viewport_test PRIVATE -UNDEBUG)
endif()
